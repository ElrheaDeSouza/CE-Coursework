1) Create a trigger for the employee table that would fire for INSERT or UPDATE operations
performed on the employee table. This trigger will display the salary difference between the
old values and new values.

//1
create or replace trigger t1
   after update or insert
   on emp_22co11
   for each row declare
   dsal number;
   begin
          dsal := :new.salary - :old.salary;
          dbms_output.put_line('Old Sal : ' || :old.salary);
          dbms_output.put_line('New Sal : ' || :new.salary);
          dbms_output.put_line('Sal Diff: ' || dsal);
   end;
   /
